⚡ Query Diagnostik Penting

── 1. Query paling lambat (top 10) ──────────────────
SELECT query, calls, mean_exec_time, total_exec_time
FROM pg_stat_statements
ORDER BY mean_exec_time DESC
LIMIT 10;

── 2. Diagnosa query spesifik ───────────────────────
EXPLAIN ANALYZE [tulis query kamu di sini];

── 3. Semua index yang ada ──────────────────────────
SELECT tablename, indexname, indexdef
FROM pg_indexes WHERE schemaname = 'public'
ORDER BY tablename;

── 4. Index tidak pernah dipakai (kandidat hapus) ───
SELECT relname, indexrelname, idx_scan
FROM pg_stat_user_indexes
WHERE idx_scan = 0 ORDER BY relname;

── 5. Ukuran tabel terbesar ─────────────────────────
SELECT relname, pg_size_pretty(pg_total_relation_size(relid))
FROM pg_stat_user_tables
ORDER BY pg_total_relation_size(relid) DESC LIMIT 10;

── 6. Cron job history ──────────────────────────────
SELECT jobname, status, start_time, return_message
FROM cron.job_run_details
ORDER BY start_time DESC LIMIT 20;

── 7. Jumlah koneksi aktif ──────────────────────────
SELECT count(*), state
FROM pg_stat_activity
GROUP BY state;

🛠️ Command Maintenance

# Push migration baru ke Supabase
supabase db push

# Kalau ada konflik history (riwayat migration tidak sinkron)
supabase migration repair --status applied [timestamp_migration]

# Buat index tanpa lock tabel (SELALU pakai ini di production)
CREATE INDEX CONCURRENTLY IF NOT EXISTS idx_[nama_deskriptif]
  ON [tabel] ([kolom]);

# Reset statistik query (setelah optimasi, agar statistik fresh)
SELECT pg_stat_statements_reset();
